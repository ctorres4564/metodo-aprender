// =====================================================================
// REGRAS DE SEGURANÇA DO FIRESTORE
// =====================================================================
// Cole este conteúdo em: Firebase Console → Firestore Database → Regras
//
// O que isso garante: cada pessoa autenticada só consegue ler e escrever
// documentos de progresso cujo ID comece com o próprio UID (o app já
// nomeia os documentos como "<uid>_<storageKey>" automaticamente).
// Ninguém consegue ler ou alterar o progresso de outra pessoa.
// =====================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /progress/{docId} {
      allow read, write: if request.auth != null
                          && docId.matches(request.auth.uid + '_.*');
    }

    // Módulos criados pelos próprios usuários dentro do app (aba "Criar novo módulo").
    // Qualquer pessoa autenticada pode criar um módulo; só quem criou (ownerId == uid)
    // pode ler, editar ou excluir depois.
    match /modules/{docId} {
      allow create: if request.auth != null
                    && request.resource.data.ownerId == request.auth.uid;
      allow read, delete: if request.auth != null
                    && resource.data.ownerId == request.auth.uid;
      // SEGURANÇA (SEC-08): ownerId é imutável no update. Sem isso, seria
      // possível "transferir" um módulo para o uid de outra pessoa,
      // injetando conteúdo arbitrário na lista "Meus Módulos" dela.
      allow update: if request.auth != null
                    && resource.data.ownerId == request.auth.uid
                    && request.resource.data.ownerId == resource.data.ownerId;
    }

    // Perfil do usuário (users/{uid}). O campo "plan" controla o limite
    // mensal de gerações por IA (ver api/_lib/usage.js) e só é alterado
    // pelo servidor (Admin SDK, que ignora estas regras) — por isso o
    // app só pode criar/editar os campos "email" e "remindersEnabled",
    // nunca "plan".
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow create: if request.auth != null && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['email', 'remindersEnabled']);
      allow update: if request.auth != null && request.auth.uid == uid
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['email', 'remindersEnabled']);
    }

    // Contador mensal de uso de IA (ai_usage/{uid}_{ano-mes}). Só leitura — quem
    // escreve é sempre o servidor (Admin SDK, que ignora estas regras). Leitura
    // é usada pra mostrar "quantas gerações você já usou este mês" na tela.
    match /ai_usage/{docId} {
      allow read: if request.auth != null && docId.matches(request.auth.uid + '_.*');
    }

    // Biblioteca de materiais (Etapa 1). O documento principal
    // (materials/{materialId}) só é escrito pelo servidor — via
    // api/material.js (ações create/updateStatus/registerModule/delete,
    // usando o Admin SDK, que ignora estas regras). O app só pode LER o
    // próprio material; isso impede que alguém manipule direto pelo
    // navegador campos administrativos como "status", "generatedModuleCount"
    // ou "processingError".
    match /materials/{materialId} {
      allow read: if request.auth != null && resource.data.ownerId == request.auth.uid;

      // O texto extraído por página já é gravado direto pelo cliente (são
      // muitos documentos pequenos por material — passar tudo por uma
      // função serverless arriscaria estourar o limite de tamanho do
      // corpo da requisição em livros grandes). A permissão é liberada só
      // para quem é dono do material "pai", verificado a cada leitura/escrita.
      match /pages/{pageId} {
        allow read, write: if request.auth != null
                            && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid;
      }

      // Leitor de PDF (Etapa 2): destaques e anotações, escritos direto pelo
      // cliente (sem função serverless — não há campo administrativo aqui,
      // diferente do documento materials/{materialId} em si). Duas camadas
      // de defesa em toda operação: (1) o próprio documento (highlight ou
      // nota) precisa ter ownerId == uid de quem está pedindo — não confia
      // só nisso, porque um campo enviado pelo cliente pode ser forjado;
      // (2) por isso TAMBÉM verifica, via get(), que o material PAI
      // realmente pertence a esse mesmo uid — impede que alguém crie um
      // highlight com seu próprio uid dentro do material de outra pessoa.
      match /highlights/{highlightId} {
        allow read: if request.auth != null
                    && resource.data.ownerId == request.auth.uid;

        allow create: if request.auth != null
                    && request.resource.data.ownerId == request.auth.uid
                    && request.resource.data.materialId == materialId
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid;

        // Só cor e posição podem mudar (trocar cor do destaque, ou -no
        // futuro- ajustar a seleção). ownerId, materialId e createdAt são
        // imutáveis; o texto destacado também (para editar o trecho, a
        // pessoa exclui o destaque e cria um novo).
        allow update: if request.auth != null
                    && resource.data.ownerId == request.auth.uid
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid
                    && request.resource.data.ownerId == resource.data.ownerId
                    && request.resource.data.materialId == resource.data.materialId
                    && request.resource.data.createdAt == resource.data.createdAt
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['color', 'position', 'updatedAt']);

        allow delete: if request.auth != null
                    && resource.data.ownerId == request.auth.uid
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid;
      }

      match /notes/{noteId} {
        allow read: if request.auth != null
                    && resource.data.ownerId == request.auth.uid;

        // Além do material pai, confere também que o highlightId informado
        // existe DENTRO deste mesmo material e pertence a este mesmo uid —
        // sem isso, seria possível criar uma nota apontando pra um
        // highlightId de outro material ou de outra pessoa (a nota ficaria
        // "solta", sem vínculo real com um trecho que a pessoa realmente
        // destacou).
        allow create: if request.auth != null
                    && request.resource.data.ownerId == request.auth.uid
                    && request.resource.data.materialId == materialId
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid
                    && get(/databases/$(database)/documents/materials/$(materialId)/highlights/$(request.resource.data.highlightId)).data.ownerId == request.auth.uid;

        // O texto e o vínculo opcional com um módulo/conceito (Etapa 3)
        // podem mudar. ownerId, materialId, highlightId (o trecho ao qual a
        // nota está vinculada) e createdAt são imutáveis.
        allow update: if request.auth != null
                    && resource.data.ownerId == request.auth.uid
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid
                    && request.resource.data.ownerId == resource.data.ownerId
                    && request.resource.data.materialId == resource.data.materialId
                    && request.resource.data.highlightId == resource.data.highlightId
                    && request.resource.data.createdAt == resource.data.createdAt
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['text', 'updatedAt', 'linkedModuleId', 'linkedConceptId']);

        allow delete: if request.auth != null
                    && resource.data.ownerId == request.auth.uid
                    && get(/databases/$(database)/documents/materials/$(materialId)).data.ownerId == request.auth.uid;
      }
    }
  }
}
