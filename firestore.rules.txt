// =====================================================================
// REGRAS DE SEGURANÇA DO FIRESTORE
// =====================================================================
// Cole este conteúdo em: Firebase Console → Firestore Database → Regras
//
// O que isso garante: cada pessoa autenticada só consegue ler e escrever
// documentos de progresso cujo ID comece com o próprio UID (o app já
// nomeia os documentos como "<uid>_<storageKey>" automaticamente).
// Ninguém consegue ler ou alterar o progresso de outra pessoa.
// =====================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /progress/{docId} {
      allow read, write: if request.auth != null
                          && docId.matches(request.auth.uid + '_.*');
    }

    // Módulos criados pelos próprios usuários dentro do app (aba "Criar novo módulo").
    // Qualquer pessoa autenticada pode criar um módulo; só quem criou (ownerId == uid)
    // pode ler, editar ou excluir depois.
    match /modules/{docId} {
      allow create: if request.auth != null
                    && request.resource.data.ownerId == request.auth.uid;
      allow read, update, delete: if request.auth != null
                    && resource.data.ownerId == request.auth.uid;
    }

    // Perfil do usuário (users/{uid}). O campo "plan" controla o limite
    // mensal de gerações por IA (ver api/_lib/usage.js) e só é alterado
    // pelo servidor (Admin SDK, que ignora estas regras) — por isso o
    // app só pode criar/editar os campos "email" e "remindersEnabled",
    // nunca "plan".
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow create: if request.auth != null && request.auth.uid == uid
                    && request.resource.data.keys().hasOnly(['email', 'remindersEnabled']);
      allow update: if request.auth != null && request.auth.uid == uid
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['email', 'remindersEnabled']);
    }

    // Contador mensal de uso de IA (ai_usage/{uid}_{ano-mes}). Só leitura — quem
    // escreve é sempre o servidor (Admin SDK, que ignora estas regras). Leitura
    // é usada pra mostrar "quantas gerações você já usou este mês" na tela.
    match /ai_usage/{docId} {
      allow read: if request.auth != null && docId.matches(request.auth.uid + '_.*');
    }
  }
}
